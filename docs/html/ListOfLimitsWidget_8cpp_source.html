33void ListOfLimitsWidget::addNewFrame(std::shared_ptr<BaseProtocol> frame)
34{
35 CollapsibleGroupBox *frameGroupBox = new CollapsibleGroupBox(this);
36 frameGroupBox->setSizePolicy(QSizePolicy::Preferred, QSizePolicy::Fixed);
37 frameGroupBox->setTitle(QString::fromStdString(frame->nameFrame));
38
39 for (auto &it : frame->orderedNames) {
40
41 std::shared_ptr<SensorData> fieldData = frame->fields[it];
42
43 SensorSettingsManager::loadSensor(it.data(), fieldData);
44
45 auto frameShared = frame;
46 QString sensorName = QString::fromStdString(it); // Локальная копия
47
48 QHBoxLayout *hBoxLt = new QHBoxLayout();
49 hBoxLt->setContentsMargins(5, 2, 5, 2);
50 hBoxLt->setSpacing(5);
51
52 QLabel *nameField = new QLabel(QString::fromStdString(it), frameGroupBox);
53 nameField->setMinimumWidth(150);
54 hBoxLt->addWidget(nameField);
55
56 // Min field
57 hBoxLt->addWidget(new QLabel("Min: ", frameGroupBox));
58 QLineEdit *minTxtEdt = new QLineEdit(frameGroupBox);
59 minTxtEdt->setValidator(new QIntValidator(INT_MIN, INT_MAX, this));
60 minTxtEdt->setText(QString::number(fieldData->limit->min));
61 minTxtEdt->setMinimumWidth(100);
62
63 connect(minTxtEdt, &QLineEdit::textChanged,
64 [sensorName, fieldData, this, minTxtEdt]() {
65 bool ok = false;
66 int value = minTxtEdt->text().toInt(&ok);
67 if (ok) {
68 fieldData->limit->min = value;
69 SensorSettingsManager::saveSensor(sensorName, fieldData);
70 }
71 });
72 hBoxLt->addWidget(minTxtEdt);
73
74 // Max field
75 hBoxLt->addWidget(new QLabel("Max: ", frameGroupBox));
76 QLineEdit *maxTxtEdt = new QLineEdit(frameGroupBox);
77 maxTxtEdt->setValidator(new QIntValidator(INT_MIN, INT_MAX, this));
78 maxTxtEdt->setText(QString::number(fieldData->limit->max));
79 maxTxtEdt->setMinimumWidth(100);
80
81 connect(maxTxtEdt, &QLineEdit::textChanged,
82 [sensorName, fieldData, this, maxTxtEdt]() {
83 bool ok = false;
84 int value = maxTxtEdt->text().toInt(&ok);
85 if (ok) {
86 fieldData->limit->max = value;
87 SensorSettingsManager::saveSensor(sensorName, fieldData);
88 }
89 });
90 hBoxLt->addWidget(maxTxtEdt);
91
92 // Checkbox
93 QCheckBox *checkBox = new QCheckBox("Детал.", frameGroupBox);
94 checkBox->setChecked(fieldData->settings->useDetalaizedLimits);
95 connect(checkBox, &QCheckBox::toggled, [fieldData, it, sensorName, this](bool checked) {
96 fieldData->settings->useDetalaizedLimits = checked;
97 SensorSettingsManager::saveSensor(sensorName, fieldData);
98 });
99 hBoxLt->addWidget(checkBox);
100
101 // Details button
102 QPushButton *openLimitsDetail = new QPushButton("...", frameGroupBox);
103 connect(openLimitsDetail, &QPushButton::clicked, [frameShared, fieldData, sensorName]() {
104 DetailingLimitsWidget *detatlsLimits = new DetailingLimitsWidget(fieldData, sensorName);
105 detatlsLimits->setAttribute(Qt::WA_DeleteOnClose);
106 detatlsLimits->show();
107 });
108 hBoxLt->addWidget(openLimitsDetail);
109
110 hBoxLt->addStretch();
111 frameGroupBox->addLayout(hBoxLt);
112 }
113
114 frameGroupBox->setExpanded(false);
