122void SmoothMotorControl::smoothControl(QPushButton *stopBtn, uint8_t num, uint16_t minPwm, uint16_t maxPwm, uint16_t timer)
123{
124 ValueIncrementer *incrementer = new ValueIncrementer(minPwm, maxPwm, timer);
125
126 if (stopBtn != nullptr)
127 connect(stopBtn, &QPushButton::clicked, [incrementer](){incrementer->stopThis();});
128
129 connect(incrementer, &ValueIncrementer::valueChanged, [num, this](double val){
130 sendMotorControlMsg(num, val);
131 });
132 connect(incrementer, &ValueIncrementer::finished, [incrementer](){
133 delete incrementer;
134 });
135
136 incrementer->start();
137}
