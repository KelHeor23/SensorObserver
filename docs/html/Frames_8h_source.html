68struct EscStatusInfo1 : BaseFrame{
69 int32_t speed = 0; // 24-bit RPM motor speed (sign-extended to 32 bits)
70 uint16_t recv_pwm = 0; // 0.1us units (little-endian)
71 uint16_t comm_pwm = 0; // 0.1us units (little-endian)
72
73 static std::shared_ptr<EscStatusInfo1> unpack(const char buffer[8]) {
74 auto result = std::make_shared<EscStatusInfo1>();
75
76 // Исправленный парсинг 24-битной скорости (little-endian)
77 uint32_t raw_speed = static_cast<uint32_t>(static_cast<uint8_t>(buffer[2]) << 16) |
78 static_cast<uint32_t>(static_cast<uint8_t>(buffer[1]) << 8) |
79 static_cast<uint8_t>(buffer[0]);
80 // Коррекция знака для 24-битного числа
81 result->speed = (raw_speed & 0x800000) ?
82 (static_cast<int32_t>(raw_speed) | 0xFF000000) :
83 raw_speed;
84
85 // Исправленный парсинг recv_pwm (little-endian)
86 result->recv_pwm = static_cast<uint16_t>(static_cast<uint8_t>(buffer[4]) << 8) |
87 static_cast<uint16_t>(static_cast<uint8_t>(buffer[3]));
88
89 // Исправленный парсинг comm_pwm (little-endian)
90 result->comm_pwm = static_cast<uint16_t>(static_cast<uint8_t>(buffer[7]) << 8) |
91 static_cast<uint16_t>(static_cast<uint8_t>(buffer[6]));
92
93 // Масштабирование PWM значений
94 result->recv_pwm /= 10;
95 result->comm_pwm /= 10;
96
97 return result;
98 }
99};

102struct EscStatusInfo2 : BaseFrame{
103 uint16_t voltage = 0; // 0.1V Bus voltage (little-endian)
104 int16_t bus_current = 0; // 0.1A Bus current (little-endian)
105 int16_t current = 0; // 0.1A Motor line current (little-endian)
106
107 static std::shared_ptr<EscStatusInfo2> unpack(const char buffer[8]) {
108 auto result = std::make_shared<EscStatusInfo2>();
109
110 // Парсинг voltage (uint16_t, little-endian)
111 result->voltage =
112 (static_cast<uint16_t>(static_cast<uint8_t>(buffer[0])) |
113 (static_cast<uint16_t>(static_cast<uint8_t>(buffer[1])) << 8)) / 10;
114
115 // Парсинг bus_current (int16_t, little-endian)
116 result->bus_current =
117 static_cast<int16_t>(
118 static_cast<uint16_t>(static_cast<uint8_t>(buffer[2])) |
119 (static_cast<uint16_t>(static_cast<uint8_t>(buffer[3])) << 8)
120 ) / 10;
121
122 // Парсинг current (int16_t, little-endian)
123 result->current =
124 static_cast<int16_t>(
125 static_cast<uint16_t>(static_cast<uint8_t>(buffer[4])) |
126 (static_cast<uint16_t>(static_cast<uint8_t>(buffer[5])) << 8)
127 ) / 10;
128
129 return result;
130 }
131};

135struct EscStatusInfo3 : BaseFrame{
136 uint8_t cap_temp = 0; // Temperature in °C (беззнаковое)
137 uint8_t mcu_temp = 0; // Temperature in °C (беззнаковое)
138 uint8_t motor_temp = 0; // Temperature in °C (беззнаковое)
139 uint16_t Error = 0;
140
141 static std::shared_ptr<EscStatusInfo3> unpack(const char buffer[8]) {
142 auto result = std::make_shared<EscStatusInfo3>();
143
144 // Парсинг температур (прямое присваивание)
145 result->cap_temp = static_cast<uint8_t>(buffer[2]) - 50;
146 result->mcu_temp = static_cast<uint8_t>(buffer[3]) - 50;
147 result->motor_temp = static_cast<uint8_t>(buffer[4]) - 50;
148
149 // Парсинг резервных байт (buffer[5] и buffer[6]) или другой running_error
150 /** \note Поле Error собирается из младшего/старшего байтов (little-endian). */
151 result->Error =
152 static_cast<uint16_t>(static_cast<uint8_t>(buffer[6])) |
153 (static_cast<uint16_t>(static_cast<uint8_t>(buffer[7])) << 8);
154
155 return result;
156 }
