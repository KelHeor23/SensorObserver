27/// \brief Выполняет пошаговый разбор TLV/потока и обновляет соответствующие виджеты.
28/// \param message Буфер входящих данных.
29void DataReader::parseMsg(const QByteArray& message)
30{
31 if (message.size() < 4) {
32 qDebug() << "Слишком короткое сообщение";
33 return;
34 }
35
36 const auto *base = reinterpret_cast<const uchar*>(message.constData());
37 const quint16 node_id = qFromLittleEndian<quint16>(base + 0);
38 const quint16 frame_id = qFromLittleEndian<quint16>(base + sizeof(quint16));
39
40 const QByteArray payloadAfterHeader = message.mid(4);
41
42 switch (frame_id) {
43 case 20022:
44 if (node_id < 32 && sensorsTableWdgt->displayngSensors.size() > int(node_id % 32))
45 sensorsTableWdgt->displayngSensors[node_id % 32]
46 ->setSensorsData(ESC_FRAME1, payloadAfterHeader.toStdString(), node_id);
47 break;
48 case 20023:
49 if (node_id < 32 && sensorsTableWdgt->displayngSensors.size() > int(node_id % 32))
50 sensorsTableWdgt->displayngSensors[node_id % 32]
51 ->setSensorsData(ESC_FRAME2, payloadAfterHeader.toStdString(), node_id);
52 break;
53 case 20024:
54 if (node_id < 32 && sensorsTableWdgt->displayngSensors.size() > int(node_id % 32))
55 sensorsTableWdgt->displayngSensors[node_id % 32]
56 ->setSensorsData(ESC_FRAME3, payloadAfterHeader.toStdString(), node_id);
57 break;
58 default:
59 break;
60 }
61
62 int it = 0;
63
64 // Должно хватать хотя бы на тег (4 байта)
65 while (it + 4 <= message.size()) {
66
67 // Читаем тег как LE-uint32
68 const quint32 tag = qFromLittleEndian<quint32>(
69 reinterpret_cast<const uchar*>(message.constData() + it)
70 );
71
72 const quint32 type = (tag & ~0b111u);
73 const quint32 idx = (tag & 0b111u);
74
75 auto ensureSpace = [&](qsizetype need) -> bool {
76 if (it + need > message.size()) {
77 qDebug() << "Ошибка чтения пакета данных (не хватает байт). type =" << type;
78 return false;
79 }
80 return true;
81 };
82
83 switch (type) {
84 case Protocol_numbers::ENGINE_SENSORS: {
85 if (!ensureSpace(sizeof(EngineSensorsData))) return;
86 const QByteArray chunk = message.mid(it, sizeof(EngineSensorsData));
87 sensorsTableWdgt->readEngineSensorsMsg(idx, chunk);
88 it += sizeof(EngineSensorsData);
89 break;
90 }
91 case Protocol_numbers::VOLTAGE_REGULATORS: {
92 if (!ensureSpace(sizeof(VoltageRegulatorsData))) return;
93 const QByteArray chunk = message.mid(it, sizeof(VoltageRegulatorsData));
94 sensorsTableWdgt->readVoltageRegulatorsMsg(idx, chunk);
95 it += sizeof(VoltageRegulatorsData);
96 break;
97 }
98 case Protocol_numbers::OTHER_SENSROS: { // если это действительно такое имя
99 if (!ensureSpace(sizeof(OtherSensorsData))) return;
100 const QByteArray chunk = message.mid(it, sizeof(OtherSensorsData));
101 sensorsTableWdgt->readOtherSensorsMsg(idx, chunk);
102 it += sizeof(OtherSensorsData);
103 break;
104 }
105 default:
106 qDebug() << "Неизвестный тип тега:" << type << " — прекращаю парсинг";
107 return;
108 }
109 }
110}
