18DetailingLimitsWidget::DetailingLimitsWidget(std::shared_ptr<SensorData> data, QString name, QWidget *parent)
19 : QWidget(parent)
20 , data_t(data)
21 , sensorName(name)
22 , mainLt(new QVBoxLayout(this)) // Установка родителя для layout
23 , fieldsLt(new QVBoxLayout())
24 , buttonsLt(new QHBoxLayout())
25 , addNewMinMaxBtn(new QPushButton("Добавить MinMax", this))
26 , addNewPointBtn(new QPushButton("Добавить точку", this))
27{
28 setWindowTitle("Поле " + sensorName);
29 buttonsLt->addWidget(addNewMinMaxBtn);
30 buttonsLt->addWidget(addNewPointBtn);
31
32 // Добавление новых элементов по индексу
33 connect(addNewMinMaxBtn, &QPushButton::clicked, this, [this](){

36 addNewMinMax(data_t->detalaizedLimits->size() - 1);
37 });
38
39 connect(addNewPointBtn, &QPushButton::clicked, this, [this](){
40 data_t->detalaizedLimits->push_back(SensorLimitsColored{{0, 0},

42 addNewPoint(data_t->detalaizedLimits->size() - 1);
43 });
44
45 // Итерация по индексам
46 for (size_t i = 0; i < data_t->detalaizedLimits->size(); ++i) {
47 const auto& it = (*data_t->detalaizedLimits)[i];
48 if (it.limit.min == it.limit.max)
49 addNewPoint(i);
50 else
51 addNewMinMax(i);
52 }
53
54 mainLt->addLayout(fieldsLt);
55 mainLt->addLayout(buttonsLt);
