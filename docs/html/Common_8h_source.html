15/**
16 * \brief Меняет порядок байт (эндивность) значения любого тривиального целочисленного типа.
17 * \tparam T Целочисленный тип (например, uint16_t, uint32_t, uint64_t).
18 * \param value Входное значение в исходной эндивности.
19 * \return Значение с обратным порядком байт.
20 * \warning Шаблон рассчитан на целочисленные типы и не предназначен для float/double.
21 */
22template <typename T>
23T swapEndianness(T value) {
24 static_assert(std::is_integral_v<T> || std::is_enum_v<T>, "swapEndianness expects integral type");
25 T result{};
26 auto* in = reinterpret_cast<unsigned char*>(&value);
27 auto* out = reinterpret_cast<unsigned char*>(&result);
28 for (std::size_t i = 0; i < sizeof(T); ++i) {
29 out[i] = in[sizeof(T)-1-i];
30 }
31 return result;
32}
