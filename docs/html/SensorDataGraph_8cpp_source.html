91void SensorDataGraph::fillSensorsList()
92{
93 sensorsListTreeWdgt->setHeaderHidden(true); // Скрываем заголовок
94 sensorsListTreeWdgt->setMinimumWidth(200);
95 sensorsListTreeWdgt->setMaximumWidth(300);
96
97 // Пример данных: группы и элементы
98 QList<QPair<QString, QStringList>> groups;
99
100 for (auto &it : sensorsManager->getFrames()) {
101 QPair<QString, QStringList> temp;
102 temp.first = QString::fromStdString(it.second->getNameFrame());
103 for (auto &field : it.second->getFields())
104 temp.second.push_back(QString::fromStdString(field.first));
105 groups.push_back(temp);
106 }
107
108 // Заполняем дерево
109 for (const auto &group : groups) {
110 // Создаем группу (родительский элемент)
111 QTreeWidgetItem *groupItem = new QTreeWidgetItem(sensorsListTreeWdgt);
112 groupItem->setText(0, group.first);
113 groupItem->setFlags(Qt::ItemIsEnabled); // Группа без чекбокса
114
115 // Добавляем элементы в группу
116 for (const QString &item : group.second) {
117 QTreeWidgetItem *childItem = new QTreeWidgetItem(groupItem);
118 childItem->setFlags(Qt::ItemIsUserCheckable | Qt::ItemIsEnabled | Qt::ItemIsSelectable);
119 childItem->setCheckState(0, Qt::Unchecked);
120 childItem->setText(0, item);
121 }
122 // Разворачиваем группу по умолчанию
123 groupItem->setExpanded(false);
124 }
125
126 // Обработка изменений состояния чекбоксов
127 QObject::connect(sensorsListTreeWdgt, &QTreeWidget::itemChanged, this, &SensorDataGraph::onItemChanged);
128}

138void SensorDataGraph::onItemChanged(QTreeWidgetItem *item, int column)
139{
140 if (column != 0) return;
141
142 QString sensorName = item->text(0);
143 QString groupName = item->parent()->text(0);
144
145 DataFullName frame;
146 frame.engineNum = 0;
147 frame.frameName = groupName;
148 frame.sensorName = sensorName;
149
150 bool isChecked = (item->checkState(0) == Qt::Checked);
151
152 if (isChecked) {
153 QCPGraph *graph = m_plot->addGraph();
154 graph->setLineStyle(QCPGraph::lsLine);
155 graph->setAdaptiveSampling(true); // Включите адаптивную выборку
156 graph->setName(sensorName);
157
158 QString parentText = item->parent()->text(0);
159 graphMap[sensorName] = graph;
160
161 // Настраиваем внешний вид
162 QColor color(getColor());
163 graph->setPen(QPen(color, 2));
164 graph->setName(sensorName);
165
166 auto &time = DataStructure::Instance().engines[0][groupName.toStdString()]["Time"];
167 auto &data = DataStructure::Instance().engines[0][groupName.toStdString()][sensorName.toStdString()];
168
169 graph->setData(QVector<double>(time.begin(), time.end()), QVector<double>(data.begin(), data.end()));
170
171 dataSize[frame] = DataStructure::Instance().engines[0][groupName.toStdString()]["Time"].size();
172 } else {
173 if (graphMap.contains(sensorName)) {
174 dataSize.erase(frame);
175 m_plot->removeGraph(graphMap[sensorName]);
176 graphMap.remove(sensorName);
177 }
178 }
179}

181void SensorDataGraph::addNewData()
182{
183 for (auto &it : dataSize) {
184 auto &data = DataStructure::Instance().engines[it.first.engineNum][it.first.frameName.toStdString()][it.first.sensorName.toStdString()];
185 if (data.size() > it.second) {
186 auto &graph = graphMap[it.first.sensorName];
187 for (int i = it.second; i < data.size(); i++){
188 auto &time = DataStructure::Instance().engines[0][it.first.frameName.toStdString()]["Time"];
189 auto &data = DataStructure::Instance().engines[0][it.first.frameName.toStdString()][it.first.sensorName.toStdString()];
190
191 graph->setData(QVector<double>(time.begin(), time.end()), QVector<double>(data.begin(), data.end()));
192 }
193 it.second = data.size();
194 }
195 }

204void SensorDataGraph::updateGraph()
205{
206 int time = 20;
207 double now = QDateTime::currentDateTime().toMSecsSinceEpoch() / 1000.;
208 m_plot->xAxis->setRange(now - time, now);
209
210 // Ограничиваем количество точек для каждого графика
211 for (auto it = graphMap.begin(); it != graphMap.end(); ++it) {
212 QCPGraph* graph = it.value();
213 if (graph->data()->size() > 5000) { // Максимум 5000 точек на график
214 graph->data()->removeBefore(now - time); // Удаляем старые данные
215 }
216 }
217 // Масштабируем только ось Y по данным
218 m_plot->yAxis->rescale();
219 m_plot->replot(QCustomPlot::rpQueuedReplot);
220}
