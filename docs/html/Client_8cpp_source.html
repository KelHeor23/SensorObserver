37void Client::connectToServer()
38{
39 static int i = 0;
40 if (reconnectTimer->isActive()) {
41 reconnectTimer->stop(); // Остановить таймер перед новой попыткой
42 }
43
44 socket->connectToHost(QHostAddress(droneIP), dronePort); // Замените на адрес и порт вашего сервера
45 if (socket->waitForConnected(1000)) {
46 qDebug() << "Connected to server";
47 reconnectTimer->stop(); // Остановить таймер, если подключение успешно
48 } else {
49 qDebug() << "Connection failed. Trying again..." << i++;
50 reconnectTimer->start(); // Запустить таймер для повторных попыток
51 }
52}
