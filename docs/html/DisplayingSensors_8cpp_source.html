46void DisplayingSensors::checkRangeValues(ColorProgressBar *pb, std::shared_ptr<SensorData> field){
47 const int value = field->val; // Кэшируем значение
48 const auto& limits = *field->detalaizedLimits; // Ссылка на вектор
49
50 pb->setMinimum(field->limit->min);
51 pb->setMaximum(field->limit->max);
52
53 // 1. Проверка детализированных лимитов (если включены)
54 if (field->settings->useDetalaizedLimits) {
55 for (const auto& it : limits) {
56 if (value >= it.limit.min && value <= it.limit.max) {
57 pb->setColor(it.color);
58 return; // Выходим сразу после установки цвета
59 }
60 }
61 }
62
63 // 2. Проверка основных лимитов (только если min < max)
64 const auto& main_limit = *field->limit;
65 if (main_limit.min >= main_limit.max) {
66 pb->setColor(Qt::transparent);
67 return;
68 }
69
70 // Кэшируем вычисляемые значения
71 const int range = main_limit.max - main_limit.min;
72 const double threshold = 0.15 * range;
73 const int low_threshold = main_limit.min + static_cast<int>(threshold);
74 const int high_threshold = main_limit.max - static_cast<int>(threshold);
75
76 // Определение цвета по основным лимитам
77 if (value < main_limit.min || value > main_limit.max) {
78 pb->setColor(Qt::red);
79 }
80 else if (value <= low_threshold || value >= high_threshold) {
81 pb->setColor(Qt::yellow);
82 }
83 else {
84 pb->setColor(Qt::green);
85 }
86}

103void DisplayingSensors::linkFrame(FrameTypes type, std::shared_ptr<SensorsFrames>target)
104{
105 auto& srcFrame = sensorManager->getFrames()[type];
106 auto& dstFrame = target->getFrames()[type];
107
108 if (!srcFrame || !dstFrame) return;
109
110 for (auto& [name, data] : srcFrame->getFields()) {
111 if (auto it = dstFrame->getFields().find(name); it != dstFrame->getFields().end()) {
112 data->linksSensorData(it->second);
113 }
114 }
115}

117void DisplayingSensors::setSensorsData(FrameTypes type, std::string_view data, int16_t node_id)
118{
119 auto& managerFrames = sensorManager->getFrames();
120 managerFrames[type]->setData(data, node_id % 32);
121 auto &fields = managerFrames[type]->getFields();
122
123 for (auto &it : managerFrames[type]->orderedNames) {
124 sensorsColorProgressBarDataLabels[it.data()]->setValue(fields[it.data()]->val);
125 checkRangeValues(sensorsColorProgressBarDataLabels[it.data()], fields[it.data()]);
126 }
127}
